## PC

```base
filters:
  and:
    - file.inFolder("母舰 Mothership/梯度下降 GD/PC")
properties:
  file.name:
    displayName: 名称
  note.player:
    displayName: 玩家
  note.class:
    displayName: 职类
  note.stress:
    displayName: 压力
  note.bends:
    displayName: 潜压
views:
  - type: table
    name: PC
    order:
      - file.name
      - player
      - class
      - stress
      - bends
      - status
      - active
    sort:
      - property: active
        direction: DESC
      - property: status
        direction: ASC

```
